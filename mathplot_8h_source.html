515 bool operator==(const mpFloatRect&) const = default;

517 // We compare with an epsilon precision
518 // NOTE: should be unnecessary as we are looking for any changes; normally this will be an exact match or a real change...
519 bool operator==(const mpFloatRect& rect) const

522 return std::fabs(a - b) < MP_EPSILON;
